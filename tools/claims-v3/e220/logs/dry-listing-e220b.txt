 297  def cmd_sync(root):
 298      root = os.path.abspath(root)
 299      repo_real = os.path.realpath(REPO)
 300      if os.path.realpath(root) == repo_real or os.path.realpath(root).startswith(repo_real + os.sep):
 301          print("SYNC REFUSED: the box root is inside the lab checkout")
 302          return 3
 303      if os.path.basename(root).startswith("ee-") or "/ee-" in os.path.realpath(root):
 304          print("SYNC REFUSED: a folder named ee-* is the model session's")
 305          return 3
 306      h = json.load(open(os.path.join(HERE, "harness-e220b.json")))
 307      os.makedirs(os.path.join(root, "run"), exist_ok=True)
 308      lk = open(os.path.join(root, "run", "box.lock"), "a+")
 309      try:
 310          fcntl.flock(lk.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
 311      except OSError:
 312          print("SYNC REFUSED: run/box.lock is held (a session, a build or a dry pass of the kit runs in this root)")
 313          return 3
 314      bad = []
 315      for row in h["files"]:
 316          src = rp(row["from"])
 317          if not os.path.isfile(src) or sha(src) != row["sha256"]:
 318              bad.append(f"{row['from']}: not at its sha256 in this checkout")
 319      if bad:
 320          print("SYNC REFUSED: " + "; ".join(bad))
 321          return 3
 322      want = {row["dest"] for row in h["files"]}
 323      n_new = n_same = 0
 324      for row in h["files"]:
 325          dst = os.path.join(root, row["dest"])
 326          os.makedirs(os.path.dirname(dst), exist_ok=True)
 327          if os.path.isfile(dst) and not os.path.islink(dst) and sha(dst) == row["sha256"]:
 328              n_same += 1
 329              continue
 330          tmp = dst + ".sync-tmp"
 331          shutil.copyfile(rp(row["from"]), tmp)
 332          os.chmod(tmp, 0o644)
 333          os.replace(tmp, dst)
 334          n_new += 1
 335      pruned = []
 336      for top in ("src", "fwgate"):
 337          for dp, dn, fn in os.walk(os.path.join(root, top)):
 338              dn[:] = [d for d in dn if d != "__pycache__"]
 339              for n in fn:
 340                  p = os.path.join(dp, n)
 341                  r = os.path.relpath(p, root)
 342                  if r not in want and not os.path.islink(p) and os.path.isfile(p):
 343                      os.remove(p)
 344                      pruned.append(r)
 345      for row in h["files"]:
 346          dst = os.path.join(root, row["dest"])
 347          got = sha(dst) if os.path.isfile(dst) and not os.path.islink(dst) else "missing"
 348          if got != row["sha256"]:
 349              bad.append(f"{row['dest']}: {got[:16]} after the copy, want {row['sha256'][:16]}")
 350      try:
 351          head = subprocess.run(["git", "-C", REPO, "rev-parse", "HEAD"], capture_output=True, text=True, timeout=30).stdout.strip()
 352      except (OSError, subprocess.TimeoutExpired):
 353          head = None
 354      rec = {"experiment": "E220", "synced": time.strftime("%Y-%m-%d %H:%M:%S %Z"), "from_head": head,
 355             "harness_sha256": sha(os.path.join(HERE, "harness-e220b.json")), "files": len(want), "copied": n_new, "already": n_same,
 356             "pruned": pruned, "verdict": "SYNCED" if not bad else "SYNC FAILED", "problems": bad}
 357      with open(os.path.join(root, "SYNC-e220b.json"), "w") as f:
 358          json.dump(rec, f, indent=1, sort_keys=True)
 359      print(f"{rec['verdict']}: {len(want)} files at the harness's sha256 ({n_new} copied, {n_same} already there, {len(pruned)} pruned "
 360            f"{pruned[:6]}) from HEAD {head}" + (f"; {'; '.join(bad)}" if bad else ""))
 361      return 0 if not bad else 14

  59  def queue_state(p):
  60      """-> ("dry" | "never" | "end" | "none", t_close in seconds or None)"""
  61      try:
