  60      h = hashlib.sha256()
  61      with open(p, "rb") as f:

 138  def h4n(S, R, P, scored, off, seed="E209|bootstrap|H4n"):
 139      """H4 re-read on J over idle (W at a watts point) divided by (1 + off): the scorer's rows_energy for H4, then its h2h"""
 140      rows = []
 141      for pid, x in scored.get("points", {}).items():
 142          if x.get("n_energy", 0) == 0:
 143              continue
 144          fam = P[pid]["family"]
 145          v2s, v1 = x.get("v2_e"), x.get("v1_e")
 146          ot = ((v1 or {}).get("J_given"), (v1 or {}).get("W_given")) if v1 else None
 147          if not v2s or not ot or ot[0] is None:
 148              rows.append({"id": pid, "d": None, "watts": x["watts"], "family": fam})
 149              continue
 150          if x["watts"]:
 151              mW = x["meas"]["W"] / (1.0 + off)
 152              rows.append({"id": pid, "d": abs(mW - ot[1]) - abs(mW - v2s["W_given"]), "watts": True, "family": fam})
 153          else:
 154              m = x["meas"]["J_over"] / (1.0 + off)
 155              a, b = S.ln_ratio(m, v2s["J_given"]), S.ln_ratio(m, ot[0])
 156              rows.append({"id": pid, "d": (abs(b) - abs(a)) if (a is not None and b is not None) else None, "watts": False, "family": fam})
 157      H = R["head_to_head"]
 158      return S.h2h(rows, H["tie_ln"]["joules"], H["tie_w"], "MM1-v1", "H4n", H["min_untied"], H["alpha"], H["rules"]["H4"]["decision"], seed)
