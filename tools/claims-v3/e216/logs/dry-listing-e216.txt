 263  def cmd_sync(root):
 264      root = os.path.abspath(root)
 265      repo_real = os.path.realpath(REPO)
 266      if os.path.realpath(root) == repo_real or os.path.realpath(root).startswith(repo_real + os.sep):
 267          print("SYNC REFUSED: the box root is inside the lab checkout")
 268          return 3
 269      if os.path.basename(root).startswith("ee-") or "/ee-" in os.path.realpath(root):
 270          print("SYNC REFUSED: a folder named ee-* is the model session's")
 271          return 3
 272      h = json.load(open(os.path.join(HERE, "harness-e216.json")))
 273      os.makedirs(os.path.join(root, "run"), exist_ok=True)
 274      lk = open(os.path.join(root, "run", "box.lock"), "a+")
 275      try:
 276          fcntl.flock(lk.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
 277      except OSError:
 278          print("SYNC REFUSED: run/box.lock is held (a session, a build or a dry pass of the kit runs in this root)")
 279          return 3
 280      bad = []
 281      for row in h["files"]:
 282          src = rp(row["from"])
 283          if not os.path.isfile(src) or sha(src) != row["sha256"]:
 284              bad.append(f"{row['from']}: not at its sha256 in this checkout")
 285      if bad:
 286          print("SYNC REFUSED: " + "; ".join(bad))
 287          return 3
 288      want = {row["dest"] for row in h["files"]}
 289      n_new = n_same = 0
 290      for row in h["files"]:
 291          dst = os.path.join(root, row["dest"])
 292          os.makedirs(os.path.dirname(dst), exist_ok=True)
 293          if os.path.isfile(dst) and not os.path.islink(dst) and sha(dst) == row["sha256"]:
 294              n_same += 1
 295              continue
 296          tmp = dst + ".sync-tmp"
 297          shutil.copyfile(rp(row["from"]), tmp)
 298          os.chmod(tmp, 0o644)
 299          os.replace(tmp, dst)
 300          n_new += 1
 301      pruned = []
 302      for top in ("src", "fwgate"):
 303          for dp, dn, fn in os.walk(os.path.join(root, top)):
 304              dn[:] = [d for d in dn if d != "__pycache__"]
 305              for n in fn:
 306                  p = os.path.join(dp, n)
 307                  r = os.path.relpath(p, root)
 308                  if r not in want and not os.path.islink(p) and os.path.isfile(p):
 309                      os.remove(p)
 310                      pruned.append(r)
 311      for row in h["files"]:
 312          dst = os.path.join(root, row["dest"])
 313          got = sha(dst) if os.path.isfile(dst) and not os.path.islink(dst) else "missing"
 314          if got != row["sha256"]:
 315              bad.append(f"{row['dest']}: {got[:16]} after the copy, want {row['sha256'][:16]}")
 316      try:
 317          head = subprocess.run(["git", "-C", REPO, "rev-parse", "HEAD"], capture_output=True, text=True, timeout=30).stdout.strip()
 318      except (OSError, subprocess.TimeoutExpired):
 319          head = None
 320      rec = {"experiment": "E216", "synced": time.strftime("%Y-%m-%d %H:%M:%S %Z"), "from_head": head,
 321             "harness_sha256": sha(os.path.join(HERE, "harness-e216.json")), "files": len(want), "copied": n_new, "already": n_same,
 322             "pruned": pruned, "verdict": "SYNCED" if not bad else "SYNC FAILED", "problems": bad}
 323      with open(os.path.join(root, "SYNC-e216.json"), "w") as f:
 324          json.dump(rec, f, indent=1, sort_keys=True)
 325      print(f"{rec['verdict']}: {len(want)} files at the harness's sha256 ({n_new} copied, {n_same} already there, {len(pruned)} pruned "
 326            f"{pruned[:6]}) from HEAD {head}" + (f"; {'; '.join(bad)}" if bad else ""))
 327      return 0 if not bad else 14
